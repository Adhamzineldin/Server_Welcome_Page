server {
    listen 443 ssl;
    server_name maayn.ddns.net maayn.tplinkdns.com;

    # SSL configuration
    ssl_certificate /etc/letsencrypt/live/maayn.ddns.net/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/maayn.ddns.net/privkey.pem; # managed by Certbot

    error_log /var/log/nginx/error.log debug;  # Enable detailed logging

    location / {
        set $port '';
        add_header Set-Cookie "test_cookie=test_value; Path=/; SameSite=None; Secure";

        # Use regex to extract the port
        if ($http_referer ~* /app(\d+)/) {
            set $port $1;
            add_header Set-Cookie "app_port=$port; Path=/; SameSite=None; Secure";
        }

        # Check for the port in the cookie
        if ($http_cookie ~* app_port=(\d+)) {
            set $port $1;
        }

        content_by_lua_block {
            local port = ngx.var.port
            if not port or port == "" then
                ngx.exec("@welcome")
                return
            end

            -- Health checks with error handling
            local res1 = ngx.location.capture("/health" .. port .. "/health_check_upstream1")
            if res1.status < 500 then
                ngx.exec("@upstream1")
            else
                local res2 = ngx.location.capture("/health" .. port .. "/health_check_upstream2")
                if res2.status < 500 then
                    ngx.exec("@upstream2")
                else
                    ngx.exec("@welcome")
                end
            end
        }

        proxy_next_upstream error timeout http_502 http_503 http_504;
        proxy_next_upstream_tries 1;

        error_page 500 502 503 504 = @fallback;
    }

    location ~ ^/app(\d+)/ {
        set $port $1;

        # This will set the upstream based on the port
        set $upstream "http://127.0.0.1:$port";

        if ($request_uri !~* "/api/") {
            add_header Set-Cookie "app_port=$port; Path=/; SameSite=None; Secure";
        }

        # Use the rewrite directive correctly to strip /app31000/
        rewrite ^/app\d+/(.*)$ /$1 break;

        # Forward the request to the correct upstream service
        proxy_pass $upstream;

        # Add necessary headers
        proxy_set_header Host $host;
        proxy_set_header Connection "";
        proxy_http_version 1.1;

        # Health checks with error handling
        content_by_lua_block {
            local port = ngx.var.port
            local res1 = ngx.location.capture("/health" .. port .. "/health_check_upstream1")
            if res1.status < 500 then
                ngx.exec("@upstream1")
            else
                local res2 = ngx.location.capture("/health" .. port .. "/health_check_upstream2")
                if res2.status < 500 then
                    ngx.exec("@upstream2")
                else
                    ngx.exec("@welcome")
                end
            end
        }
    }

    location @upstream1 {
        if ($request_uri !~* "/api/") {
            add_header Set-Cookie "app_port=$port; Path=/; SameSite=None; Secure";
        }

        # This rewrite should not be necessary since we already handle it in the previous location block.
        # rewrite ^/app\d+/(.*)$ /$1 break;

        proxy_pass http://127.0.0.1:$port;
        proxy_next_upstream error timeout http_502 http_503 http_504;
        proxy_next_upstream_tries 1;
    }

    location @upstream2 {
        if ($request_uri !~* "/api/") {
            add_header Set-Cookie "app_port=$port; Path=/; SameSite=None; Secure";
        }

        # This rewrite should not be necessary since we already handle it in the previous location block.
        # rewrite ^/app\d+/(.*)$ /$1 break;

        proxy_pass http://192.168.0.125:$port;
        proxy_connect_timeout 5s;
        proxy_read_timeout 5s;
        proxy_next_upstream error timeout http_502 http_503 http_504;
        proxy_next_upstream_tries 1;
    }

    location @welcome {
        root /home/adham-zineldin/server/Server_Welcome_Page;
        index index.html;
    }

    location ~ ^/health(\d+)/health_check_upstream1 {
        set $port $1;
        proxy_connect_timeout 2s;
        proxy_read_timeout 2s;
        proxy_pass http://127.0.0.1:$port/health_check;

        proxy_http_version 1.1;
    }

    location ~ ^/health(\d+)/health_check_upstream2 {
        set $port $1;
        proxy_connect_timeout 1s;
        proxy_read_timeout 1s;
        proxy_pass http://192.168.0.125:$port/health_check;
        proxy_set_header Host $host;
        proxy_set_header Connection "";
        proxy_http_version 1.1;
    }

    error_page 502 503 504 = /custom_502.html;
